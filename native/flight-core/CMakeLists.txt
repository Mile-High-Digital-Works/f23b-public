cmake_minimum_required(VERSION 3.20)
project(f23b_flight_core LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT FIT_PARAMETERS_DIR)
  message(FATAL_ERROR "Supply FIT_PARAMETERS_DIR with the recorded fit header")
endif()
add_executable(flight_core_replay replay.cpp)
target_include_directories(flight_core_replay PRIVATE "${FIT_PARAMETERS_DIR}")
add_library(flight_core_offline SHARED offline_api.cpp)
target_include_directories(flight_core_offline PRIVATE "${FIT_PARAMETERS_DIR}")
if(INSTALLED_CORE_LIBRARY)
  add_executable(installed_core_replay replay.cpp)
  target_compile_definitions(installed_core_replay PRIVATE INSTALLED_CONTROL)
  target_compile_features(installed_core_replay PRIVATE cxx_std_20)
  target_include_directories(installed_core_replay PRIVATE "${FIT_PARAMETERS_DIR}" "../flight-feel/src/efm/performance/include")
  target_link_libraries(installed_core_replay PRIVATE "${INSTALLED_CORE_LIBRARY}")
endif()
enable_testing()
add_executable(flight_core_unit core_test.cpp)
target_include_directories(flight_core_unit PRIVATE "${FIT_PARAMETERS_DIR}")
add_test(NAME flight_core_pure_functions COMMAND flight_core_unit)
add_executable(flight_core_tuning_test tuning_test.cpp)
target_include_directories(flight_core_tuning_test PRIVATE "${FIT_PARAMETERS_DIR}")
add_test(NAME flight_core_tuning COMMAND flight_core_tuning_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME flight_core_segment_boundaries COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/calibration_replay_test.py")
if(MSVC)
  target_compile_options(flight_core_replay PRIVATE /W4 /WX)
  target_compile_options(flight_core_offline PRIVATE /W4 /WX)
  target_compile_options(flight_core_unit PRIVATE /W4 /WX)
else()
  target_compile_options(flight_core_replay PRIVATE -Wall -Wextra -Werror)
  target_compile_options(flight_core_unit PRIVATE -Wall -Wextra -Werror)
endif()
