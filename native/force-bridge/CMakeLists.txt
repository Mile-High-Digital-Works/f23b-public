cmake_minimum_required(VERSION 3.25)
project(F23B_NativeForceBridge LANGUAGES CXX ASM_MASM)
include(CTest)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DCS_ROOT "D:/Eagle Dynamics/DCS World" CACHE PATH "Installed DCS root")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_forwarding.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/native_exports.txt" "${CMAKE_CURRENT_BINARY_DIR}"
  COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
  "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}"
  COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory(../flight-feel flight-feel)
add_library(F23B_ForceBridge SHARED bridge.cpp "${CMAKE_CURRENT_BINARY_DIR}/loader.cpp"
  "${CMAKE_CURRENT_BINARY_DIR}/forward.asm" "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
target_include_directories(F23B_ForceBridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${DCS_ROOT}/API/include")
target_compile_definitions(F23B_ForceBridge PRIVATE NOMINMAX)
target_link_libraries(F23B_ForceBridge PRIVATE f23b_performance_core)
target_compile_features(F23B_ForceBridge PRIVATE cxx_std_20)
if(BUILD_TESTING)
  add_executable(f23b_ground_yaw_tests ground_yaw_tests.cpp)
  target_link_libraries(f23b_ground_yaw_tests PRIVATE f23b_performance_core)
  target_compile_features(f23b_ground_yaw_tests PRIVATE cxx_std_20)
  add_test(NAME f23b_ground_yaw_tests COMMAND f23b_ground_yaw_tests)
  add_executable(f23b_native_control_output_tests control_output_tests.cpp)
  target_include_directories(f23b_native_control_output_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${DCS_ROOT}/API/include")
  target_compile_definitions(f23b_native_control_output_tests PRIVATE NOMINMAX)
  target_link_libraries(f23b_native_control_output_tests PRIVATE f23b_performance_core)
  target_compile_features(f23b_native_control_output_tests PRIVATE cxx_std_20)
  add_test(NAME f23b_native_control_output_tests COMMAND f23b_native_control_output_tests)
  add_executable(f23b_native_propulsion_tests propulsion_tests.cpp)
  target_link_libraries(f23b_native_propulsion_tests PRIVATE f23b_performance_core)
  target_compile_features(f23b_native_propulsion_tests PRIVATE cxx_std_20)
  add_test(NAME f23b_native_propulsion_tests COMMAND f23b_native_propulsion_tests)
  add_executable(f23b_native_steering_tests steering_tests.cpp)
  target_compile_features(f23b_native_steering_tests PRIVATE cxx_std_20)
  add_test(NAME f23b_native_steering_tests COMMAND f23b_native_steering_tests)
